cmake_minimum_required(VERSION 3.24)
project(OpenDeskNative VERSION 3.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Svg Widgets)
if(NOT DEFINED TURBOJPEG_ROOT)
  message(FATAL_ERROR "Set TURBOJPEG_ROOT to the pinned libjpeg-turbo 3.2.1 MSVC x64 SDK")
endif()
add_library(OpenDeskJpeg SHARED IMPORTED)
set_target_properties(OpenDeskJpeg PROPERTIES IMPORTED_IMPLIB "${TURBOJPEG_ROOT}/lib/turbojpeg.lib" IMPORTED_LOCATION "${TURBOJPEG_ROOT}/bin/turbojpeg.dll" INTERFACE_INCLUDE_DIRECTORIES "${TURBOJPEG_ROOT}/include")
if(NOT DEFINED OPENDESK_ENGINE_LIB)
  message(FATAL_ERROR "Set OPENDESK_ENGINE_LIB to the built Rust static library")
endif()
qt_standard_project_setup(REQUIRES 6.8)
configure_file(opendesk.rc.in "${CMAKE_CURRENT_BINARY_DIR}/opendesk.rc" @ONLY)
qt_add_executable(OpenDeskNative WIN32 main.cpp jpeg_codec.cpp jpeg_codec.h engine_bridge.cpp engine_bridge.h frame_item.cpp frame_item.h resources.qrc "${CMAKE_CURRENT_BINARY_DIR}/opendesk.rc")
set_target_properties(OpenDeskNative PROPERTIES OUTPUT_NAME OpenDesk)
set_source_files_properties(qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
qt_add_qml_module(OpenDeskNative URI OpenDesk.Native VERSION 1.0
  QML_FILES qml/Main.qml qml/Theme.qml qml/ActionButton.qml qml/Panel.qml qml/Glyph.qml qml/Field.qml qml/RemoteCursor.qml qml/VersionBadge.qml)
target_link_libraries(OpenDeskNative PRIVATE "${OPENDESK_ENGINE_LIB}" OpenDeskJpeg Qt6::Core Qt6::Gui Qt6::Quick Qt6::QuickControls2 Qt6::Svg Qt6::Widgets
  userenv ws2_32 crypt32 secur32 bcrypt ncrypt advapi32 iphlpapi ole32 oleaut32 uuid propsys avrt winmm d3d11 dxgi dwmapi ntdll user32 gdi32 shell32)
target_compile_definitions(OpenDeskNative PRIVATE QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
if(MSVC)
  target_compile_options(OpenDeskNative PRIVATE /W4 /utf-8)
endif()
add_custom_command(TARGET OpenDeskNative POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${TURBOJPEG_ROOT}/bin/turbojpeg.dll" "$<TARGET_FILE_DIR:OpenDeskNative>")
